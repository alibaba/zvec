cmake_minimum_required(VERSION 3.26)
if(NOT TARGET zvec)
  project(pr889_linux_bench LANGUAGES CXX)
  get_filename_component(ZVEC_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
  set(ZVEC_LIBRARY "" CACHE FILEPATH "Path to libzvec.so from the tested build")
  if(NOT EXISTS "${ZVEC_LIBRARY}")
    message(FATAL_ERROR "Set ZVEC_LIBRARY to the tested libzvec.so")
  endif()
endif()
add_executable(pr889_filter_bench "${CMAKE_CURRENT_LIST_DIR}/filter_bench.cc")
target_compile_features(pr889_filter_bench PRIVATE cxx_std_17)
target_compile_options(pr889_filter_bench PRIVATE -O3 -march=x86-64)
if(TARGET zvec)
  find_package(Threads REQUIRED)
  zvec_add_mobile_static_sdk(pr889_benchmark_static_sdk LIBS zvec zvec_core zvec_ailego zvec_turbo)
  target_link_libraries(pr889_filter_bench PRIVATE pr889_benchmark_static_sdk)
  if(ZVEC_ENABLE_SCALAR_JIT)
    target_compile_definitions(pr889_filter_bench PRIVATE PR889_INTERNAL_STATS=1 ZVEC_ENABLE_SCALAR_JIT=1)
    target_include_directories(pr889_filter_bench PRIVATE "${PROJECT_ROOT_DIR}/src" "${EXTERNAL_INC_DIR}")
  endif()
else()
  target_include_directories(pr889_filter_bench PRIVATE "${ZVEC_ROOT}/src/include")
  target_link_libraries(pr889_filter_bench PRIVATE "${ZVEC_LIBRARY}" pthread)
endif()
